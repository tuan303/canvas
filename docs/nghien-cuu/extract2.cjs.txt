#!/usr/bin/env node
/* extract2.cjs — bóc dữ liệu đề từ MỌI biến thể gói SCORM do bộ tạo đề NSHM sinh ra.
 *
 * Dùng:
 *   node extract2.cjs <thư mục gói đã giải nén | index.html> [out.json]
 *   node extract2.cjs --all <corpusDir>      (mỗi thư mục con -> <corpusDir>/<tên>.json)
 *
 * Kết quả (JSON):
 *   { source, variant, encrypted, nKeys, EXAM, PARTS, images, raw, grading, warnings }
 *   - EXAM/PARTS: CHUẨN HOÁ về lược đồ của template hiện tại (references/dang-cau-hoi.md):
 *       PART {label, sub, note, points, passage?, groups:[GROUP]}
 *       GROUP {head, instr[], type, perItem, items[], ...riêng từng dạng}
 *     mọi trường `ans` là đáp án THẬT (đã giải mã nếu gói mã hoá).
 *   - raw: các biến gốc trong gói (đã giải mã) để không mất thông tin.
 *   - images: ảnh được tham chiếu / có trong gói / thiếu / thừa.
 *
 * Các biến thể đã gặp (xem hàm detect()):
 *   tpl-enc     EXAM+LOCK+_KM+_KX+PARTS(JSON)   — template hiện tại, đáp án mã hoá (HSA 2025)
 *   tpl-plain   EXAM+LOCK+PARTS                 — template cũ hơn, chưa mã hoá/chưa trộn (K2 MD01)
 *   ielts-v0    EXAM_INFO+EXAM_LOCK+TFNG+PASSAGES — bộ máy IELTS đời đầu (K10)
 *   thpt-v0     EXAM_INFO(ptI/ptII/ptIII)+PARTS{key,type,items}+ATOM_INFO (K12 Hoá)
 *   k4-v0       EXAM_INFO+EXAM_LOCK+EXERCISES   (K4 Tiếng Anh)
 *   kid-adv     EXAM_INFO+TEACHER_PIN+PARTS{id,icon,type:imageChoice|match|letterMatch|choice|reading|yesno|order}
 *   kid-math    QUESTIONS[{type:choice|input, visual}] + function visualHTML (Toán lớp 2)
 */
"use strict";

const path = require("path");
const vm = require("vm");

const LET = ["A", "B", "C", "D", "E", "F", "G", "H"];

/* ---------------------------------------------------------------- helpers */
function esc(s) {
  return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}
function decodeEntities(s) {
  return String(s)
    .replace(/&nbsp;/g, " ").replace(/&lt;/g, "<").replace(/&gt;/g, ">")
    .replace(/&quot;/g, '"').replace(/&#39;/g, "'").replace(/&amp;/g, "&");
}
function clone(x) { return x === undefined ? undefined : JSON.parse(JSON.stringify(x)); }

/** Lấy nội dung các thẻ <script> nội tuyến. */
function inlineScripts(html) {
  const out = [];
  const re = /<script(\s[^>]*)?>([\s\S]*?)<\/script>/gi;
  let m;
  while ((m = re.exec(html))) {
    if (m[1] && /\bsrc\s*=/.test(m[1])) continue;
    out.push({ start: m.index, body: m[2] });
  }
  return out;
}

/** Vùng dữ liệu = đầu thẻ script chứa dữ liệu .. trước dòng "var SCORM =" (mọi biến thể đều có).
 *  Với template có mốc "PHẦN 1/2" … "HẾT PHẦN SỬA. PHẦN 2/2" thì dùng đúng hai mốc đó. */
function dataRegion(html) {
  const scripts = inlineScripts(html);
  const sc = scripts.find((s) => /\b(var|let|const)\s+(EXAM|EXAM_INFO|PARTS|PASSAGES|EXERCISES|QUESTIONS)\s*=/.test(s.body));
  if (!sc) throw new Error("Không tìm thấy thẻ <script> chứa dữ liệu đề");
  let body = sc.body;
  let how;
  const iMark = body.indexOf("PHẦN 1/2");
  const iEndMark = body.search(/HẾT PHẦN SỬA\.\s*PHẦN 2\/2/);
  if (iMark >= 0 && iEndMark > iMark) {
    // mốc mở nằm trong comment "/* === ►►► PHẦN 1/2 … === */": bắt đầu sau dấu */ đầu tiên
    const a = body.indexOf("*/", iMark) + 2;
    let b = body.lastIndexOf("/*", iEndMark);
    body = body.slice(a, b);
    how = "markers";
  } else {
    const m = /^[ \t]*var\s+SCORM\s*=/m.exec(body);
    if (!m) throw new Error("Không tìm thấy mốc 'var SCORM' để cắt vùng dữ liệu");
    body = body.slice(0, m.index);
    how = "before-var-SCORM";
  }
  return { src: body, how, script: sc.body };
}

/** Bóc 1 hàm khai báo `function name(...){...}` (đếm ngoặc, bỏ qua chuỗi). */
function sliceFunction(code, name) {
  const re = new RegExp("function\\s+" + name + "\\s*\\(");
  const m = re.exec(code);
  if (!m) return null;
  let i = code.indexOf("{", m.index), depth = 0, q = null;
  for (let j = i; j < code.length; j++) {
    const c = code[j];
    if (q) { if (c === "\\") { j++; continue; } if (c === q) q = null; continue; }
    if (c === '"' || c === "'" || c === "`") { q = c; continue; }
    if (c === "{") depth++;
    else if (c === "}") { depth--; if (depth === 0) return code.slice(m.index, j + 1); }
  }
  return null;
}

/** Giải bảng đáp án: base64 -> byte XOR _KM[i % len] XOR ((i*31)&0xff) -> UTF-8 JSON.
 *  (đúng như hàm DA() trong template: atob + charCodeAt + TextDecoder) */
function decryptKeys(kx, km) {
  const b = Buffer.from(kx, "base64");
  const k = Buffer.from(km, "latin1"); // template dùng km.charCodeAt(i) -> ASCII; khoá đều là ASCII
  for (let i = 0; i < b.length; i++) b[i] ^= k[i % k.length] ^ ((i * 31) & 0xff);
  return JSON.parse(b.toString("utf8"));
}

/** Thay mọi `ans` kiểu số bằng KEYS[số]. Trả về số chỗ đã thay. */
function decryptAns(node, KEYS, stat) {
  if (Array.isArray(node)) { node.forEach((x) => decryptAns(x, KEYS, stat)); return; }
  if (!node || typeof node !== "object") return;
  for (const key of Object.keys(node)) {
    const v = node[key];
    if (key === "ans" && typeof v === "number") {
      if (v < 0 || v >= KEYS.length) stat.bad.push(v);
      node[key] = KEYS[v];
      stat.used.add(v);
      stat.count++;
    } else decryptAns(v, KEYS, stat);
  }
}

function walkStrings(node, fn, p) {
  p = p || "";
  if (typeof node === "string") { fn(node, p); return; }
  if (Array.isArray(node)) { node.forEach((x, i) => walkStrings(x, fn, p + "[" + i + "]")); return; }
  if (node && typeof node === "object") for (const k of Object.keys(node)) walkStrings(node[k], fn, p + "." + k);
}

function collectImages(data, extraFields) {
  const refs = new Map(); // src -> [paths]
  const add = (src, p) => { if (!refs.has(src)) refs.set(src, []); refs.get(src).push(p); };
  walkStrings(data, (s, p) => {
    const re = /<img\b[^>]*?\bsrc\s*=\s*["']([^"']+)["']/gi;
    let m;
    while ((m = re.exec(s))) add(m[1], p);
    // các trường ảnh "trần"
    if (/\.(img|image)$/.test(p) && /\.(png|jpe?g|gif|svg|webp)$/i.test(s)) add(s, p);
  });
  return refs;
}

function listFiles(dir) {
  const out = [];
  (function rec(d, rel) {
    for (const f of fs.readdirSync(d)) {
      const fp = path.join(d, f), r = rel ? rel + "/" + f : f;
      if (fs.statSync(fp).isDirectory()) rec(fp, r); else out.push(r);
    }
  })(dir, "");
  return out;
}

/* ---------------------------------------------------------------- detect */
function detect(g) {
  if (g.EXAM && g.PARTS && Array.isArray(g.PARTS) && g.PARTS.some((p) => p.groups))
    return g._KX ? "tpl-enc" : "tpl-plain";
  if (g.PASSAGES) return "ielts-v0";
  if (g.EXERCISES) return "k4-v0";
  if (g.QUESTIONS) return "kid-math";
  if (g.EXAM_INFO && g.PARTS && g.PARTS.some((p) => p.key && p.items)) return "thpt-v0";
  if (g.EXAM_INFO && g.PARTS && g.PARTS.some((p) => p.icon !== undefined || p.id !== undefined)) return "kid-adv";
  return "unknown";
}

/* ---------------------------------------------------------------- normalisers */
const DEFAULT_LADDER = [0, 0.1, 0.25, 0.5, 1.0];

function examFromInfo(info, meta, extra) {
  return Object.assign({
    title: meta.title || "", org: meta.org || "", subtitle: meta.subtitle || "", code: meta.code || "",
    minutes: info && info.minutes != null ? info.minutes : null,
    maxScore: info ? info.maxScore : null, passScore: info ? info.passScore : null,
    pickBest: null, showAnswers: !!(info && info.showAnswers), showBand: !!(info && info.showBand),
    scormScale: (info && info.scormScale) || null, statusMode: (info && info.statusMode) || null,
    tfLadder: DEFAULT_LADDER.slice(), shuffle: null, note: "", rules: [],
  }, extra || {});
}

function normIelts(g, meta) {
  const EXAM = examFromInfo(g.EXAM_INFO, meta);
  const PARTS = g.PASSAGES.map((P) => ({
    label: "READING PASSAGE " + P.no, sub: P.title, note: "Questions " + P.range, points: "",
    passage: { head: "READING PASSAGE " + P.no, title: P.title, sub: P.sub || "", intro: P.intro || "", paras: clone(P.paras) },
    groups: P.groups.map((G) => {
      const N = clone(G);
      if (N.type === "choice") N.type = "mcq"; // choice = mcq với opts {k,t}
      N.perItem = 1;
      return N;
    }),
  }));
  return { EXAM, PARTS, grading: {
    perItem: 1, multi: "1 điểm mỗi chữ cái đúng (tối đa nums.length), không trừ chọn sai",
    textMatch: "normGap: trim, lowercase, ‘’ʼ->', “”->\", gộp khoảng trắng, bỏ dấu câu cuối; hoặc loose = bỏ '-' và khoảng trắng",
    choice: "so khớp chính xác khoá k (VD 'NOT GIVEN', 'C')" } };
}

function normThpt(g, meta) {
  const I = g.EXAM_INFO;
  const code = (meta.title.match(/Mã đề\s*(\w+)/) || [])[1] || "";
  const EXAM = examFromInfo(I, meta, { code, tfLadder: I.ptII || DEFAULT_LADDER, note: g.ATOM_INFO || "" });
  const PARTS = g.PARTS.map((P) => ({
    label: P.label, sub: P.sub, note: P.note || "", points: P.points || "",
    groups: [{
      head: "", type: P.type,
      perItem: P.type === "mcq" ? I.ptI : P.type === "short" ? I.ptIII : 1,
      items: clone(P.items),
    }],
  }));
  return { EXAM, PARTS, grading: {
    mcq: "ptI=" + I.ptI + " / câu", short: "ptIII=" + I.ptIII + " / câu", tf: "ptII bậc thang theo số ý đúng " + JSON.stringify(I.ptII),
    textMatch: "normShort: trim, lowercase, BỎ HẾT khoảng trắng, '.'->','" } };
}

function normK4(g, meta) {
  const EXAM = examFromInfo(g.EXAM_INFO, meta);
  const PARTS = [];
  let cur = null;
  g.EXERCISES.forEach((ex) => {
    if (ex.roman || !cur) { cur = { label: ex.roman || "", sub: "", note: "", points: "", groups: [] }; PARTS.push(cur); }
    const G = {
      head: ex.no + ". " + esc(ex.instr), instr: ex.sub ? [esc(ex.sub)] : [],
      perItem: 1, points: ex.points, _origType: ex.type,
    };
    if (ex.type === "bank") {
      Object.assign(G, {
        type: "bank", bank: ex.bank.slice(),
        example: ex.example ? { q: esc(ex.example.q), ans: ex.example.ans } : undefined,
        items: ex.items.map((it) => ({ n: it.n, q: esc(it.q), ans: [it.ans].concat(it.alt || []) })),
      });
    } else if (ex.type === "mistake") {
      Object.assign(G, { type: "mistake", items: ex.items.map((it) => ({ n: it.n, ans: it.ans, seg: it.seg.map((s) => (s.k ? { t: esc(s.t), k: s.k } : { t: esc(s.t) })) })) });
    } else {
      // mcq | mcq3 | cloze -> mcq (opts là chuỗi THƯỜNG, bộ máy cũ esc() khi vẽ)
      Object.assign(G, {
        type: "mcq",
        example: ex.example ? { q: esc(ex.example.q || ""), opts: ex.example.opts.map(esc), ans: ex.example.ans } : undefined,
        items: ex.items.map((it) => ({ n: it.n, q: it.q ? esc(it.q) : "", opts: it.opts.map(esc), ans: it.ans })),
      });
      if (ex.type === "cloze") { G.pre = '<div class="passage">' + ex.text + "</div>"; G.giuThuTu = true; }
    }
    if (G.example === undefined) delete G.example;
    cur.groups.push(G);
  });
  return { EXAM, PARTS, grading: {
    perItem: "1 điểm/câu (ex.points = số câu của bài)",
    bank: "norm: trim, lowercase, gộp khoảng trắng, bỏ dấu câu cuối; chấp nhận ans + alt[], và bỏ mạo từ a/an/the đầu",
    note: "q/opts/seg/instr gốc là văn bản THƯỜNG (bộ máy esc()); bản chuẩn hoá đã esc() sẵn. Riêng cloze.text là HTML." } };
}

function normKidAdv(g, meta) {
  const EXAM = examFromInfo(g.EXAM_INFO, meta);
  const PARTS = g.PARTS.map((P) => {
    const part = { label: "Chặng " + P.id + ": " + P.title, sub: P.subtitle || "", note: "", points: "", icon: P.icon, groups: [] };
    const G = { head: "", instr: [], perItem: 1, _origType: P.type };
    if (P.type === "imageChoice") {
      Object.assign(G, { type: "mcq", pics: true, cols: 3,
        items: P.items.map((it) => ({ n: it.n, q: "<b>" + esc(it.prompt) + "</b>", ans: it.ans,
          opts: it.opts.map((o) => ({ k: o.k, t: '<img src="' + o.img + '" alt="' + o.k + '">' })) })) });
    } else if (P.type === "match") {
      Object.assign(G, { type: "bank", bank: P.bank.map((b) => b.v), bankImg: P.bank.map((b) => ({ w: b.v, img: b.img })),
        exactMatch: true,
        items: P.items.map((it) => ({ n: it.n, q: esc(it.prompt), ans: [it.ans] })) });
    } else if (P.type === "letterMatch") {
      const ex = P.bank.find((b) => b.example);
      Object.assign(G, { type: "letters", keys: P.bank.map((b) => b.v), people: P.bank.map((b) => ({ k: b.v, t: esc(b.t) })),
        exampleKey: ex ? ex.v : undefined,
        items: P.items.map((it) => ({ n: it.n, q: esc(it.prompt), ans: it.ans })) });
    } else if (P.type === "choice" || P.type === "reading") {
      Object.assign(G, { type: "mcq", items: P.items.map((it) => ({ n: it.n, q: esc(it.prompt), ans: it.ans, opts: it.opts.map((o) => ({ k: o.k, t: esc(o.t) })) })) });
    } else if (P.type === "yesno") {
      Object.assign(G, { type: "mcq", opts: [{ k: "YES", t: "Yes" }, { k: "NO", t: "No" }], items: P.items.map((it) => ({ n: it.n, q: esc(it.prompt), ans: it.ans })) });
    } else if (P.type === "order") {
      Object.assign(G, { type: "order", perItem: 2, partial: "0,5 điểm cho mỗi thẻ đúng vị trí",
        items: P.items.map((it) => {
          const t = (id) => (it.chunks.find((c) => c.id === id) || {}).t;
          const sentence = it.target.map(t).join(" ").replace(/\s+([,.!?])/g, "$1");
          return { n: it.n, q: esc(it.prompt) + ' <b>(' + esc(it.cue) + ")</b>", img: it.img, cue: it.cue,
            chunks: clone(it.chunks), target: it.target.slice(), ans: sentence };
        }) });
    } else {
      Object.assign(G, { type: "UNKNOWN:" + P.type, items: clone(P.items) });
    }
    if (P.passage) {
      part.passage = { head: "", title: esc(P.readingTitle || ""), sub: "", intro: "",
        paras: [{ k: "", t: (P.image ? '<img src="' + P.image + '" alt="">' : "") + P.passage }] };
    }
    part.groups.push(G);
    return part;
  });
  return { EXAM, PARTS, grading: {
    perItem: "1 điểm; riêng order 2 điểm (0,5 × số thẻ đúng vị trí)",
    match: "so khớp CHÍNH XÁC String(got)===String(ans) (kéo thả nên không có biến thể)" } };
}

function normKidMath(g, meta, visualHTML) {
  const EXAM = examFromInfo(null, meta, { maxScore: g.QUESTIONS.length, minutes: 0 });
  const groups = g.QUESTIONS.map((q) => {
    const vis = visualHTML ? visualHTML(q) : "";
    if (q.type === "choice") {
      const idx = q.options.indexOf(q.answer);
      return { head: "", type: "mcq", perItem: 1, cols: 3, items: [{ n: q.n, q: esc(q.prompt), table: vis, visual: q.visual,
        opts: q.options.map(esc), ans: idx >= 0 ? LET[idx] : "UNMATCHED:" + q.answer, ansText: q.answer }] };
    }
    return { head: "", type: "short", perItem: 1, items: [{ n: q.n, q: esc(q.prompt), table: vis, visual: q.visual,
      ans: [q.answer], unit: q.unit || "", hint: q.unit ? "Đơn vị: " + q.unit : "" }] };
  });
  return { EXAM, PARTS: [{ label: "Phiếu bài tập", sub: "", note: "", points: "", groups }], grading: {
    perItem: 1, textMatch: "CHÍNH XÁC sau trim: got===answer (không chuẩn hoá); choice lưu GIÁ TRỊ phương án, không phải chữ cái" } };
}

/* ---------------------------------------------------------------- CSS của gói */
/** Trải phẳng các khối <style> thành [{media, sel, body}] (bỏ comment, đi vào @media). */
function parseCss(html) {
  const css = [...html.matchAll(/<style[^>]*>([\s\S]*?)<\/style>/gi)].map((m) => m[1]).join("\n").replace(/\/\*[\s\S]*?\*\//g, "");
  function parse(src, media) {
    const out = []; let i = 0;
    while (i < src.length) {
      const ob = src.indexOf("{", i); if (ob < 0) break;
      const head = src.slice(i, ob).trim();
      let d = 0, j = ob;
      for (; j < src.length; j++) { if (src[j] === "{") d++; else if (src[j] === "}") { d--; if (d === 0) break; } }
      const body = src.slice(ob + 1, j);
      if (/^@(media|supports)/.test(head)) out.push(...parse(body, head));
      else if (!head.startsWith("@")) out.push({ media, sel: head.replace(/\s+/g, " "), body: body.replace(/\s+/g, " ").trim() });
      i = j + 1;
    }
    return out;
  }
  return parse(css, "");
}
/** Luật CSS cần "inline" khi đưa sang LMS: luật nhắc tới lớp có trong nội dung đề + :root (giá trị var()). */
function cssForContent(html, data) {
  const rules = parseCss(html);
  const classes = new Set();
  walkStrings(data, (s) => {
    for (const m of s.matchAll(/\bclass\s*=\s*["']([^"']+)["']/g)) m[1].split(/\s+/).filter(Boolean).forEach((c) => classes.add(c));
  });
  const rootVars = {};
  rules.filter((r) => r.sel === ":root").forEach((r) => r.body.split(";").forEach((d) => {
    const i = d.indexOf(":"); if (i > 0 && d.trim().startsWith("--")) rootVars[d.slice(0, i).trim()] = d.slice(i + 1).trim();
  }));
  const byClass = {};
  [...classes].forEach((c) => {
    const re = new RegExp("\\." + c.replace(/[-]/g, "\\-") + "(?![\\w-])");
    byClass[c] = rules.filter((r) => re.test(r.sel) && !/body\.exam-armed|user-select/.test(r.sel + r.body))
      .map((r) => (r.media ? "[" + r.media + "] " : "") + r.sel + " { " + r.body + " }");
  });
  const tagRules = rules.filter((r) => !r.media && /^(math|math mo\.hemo|sub|sup|table\.mini.*)$/.test(r.sel)).map((r) => r.sel + " { " + r.body + " }");
  return { classes: byClass, tags: tagRules, rootVars };
}

/* ---------------------------------------------------------------- zip input */
/** Giải nén .zip vào thư mục tạm (dùng `unzip` nếu có, nếu không thì PowerShell Expand-Archive). */
function unzipToTemp(zipPath) {
  const cp = require("child_process"), os = require("os");
  const out = fs.mkdtempSync(path.join(os.tmpdir(), "scorm-"));
  try { cp.execFileSync("unzip", ["-o", "-q", zipPath, "-d", out], { stdio: "ignore" }); }
  catch (e) {
    cp.execFileSync("powershell", ["-NoProfile", "-Command",
      "Expand-Archive -LiteralPath '" + zipPath.replace(/'/g, "''") + "' -DestinationPath '" + out.replace(/'/g, "''") + "' -Force"], { stdio: "ignore" });
  }
  return out;
}

function redactPins(g) {
  ["LOCK", "EXAM_LOCK"].forEach((k) => { if (g[k] && g[k].pin) g[k] = Object.assign({}, g[k], { pin: "***", pinHash: g[k].pinHash ? "***" : "" }); });
  if (g.TEACHER_PIN) g.TEACHER_PIN = "***";
  return g;
}

/* ---------------------------------------------------------------- main extract */
function extract(dirOrFile) {
  if (/\.zip$/i.test(dirOrFile)) {
    const r = extract(unzipToTemp(dirOrFile));
    r.source = path.basename(dirOrFile, path.extname(dirOrFile));
    return r;
  }
  let dir = dirOrFile, htmlPath = dirOrFile;
  if (fs.statSync(dirOrFile).isDirectory()) htmlPath = path.join(dirOrFile, "index.html");
  else dir = path.dirname(dirOrFile);
  const html = fs.readFileSync(htmlPath, "utf8");
  const warnings = [];

  const meta = {};
  meta.htmlTitle = decodeEntities(((html.match(/<title>([\s\S]*?)<\/title>/i) || [])[1] || "").trim());
  try {
    const man = fs.readFileSync(path.join(dir, "imsmanifest.xml"), "utf8");
    const t = [...man.matchAll(/<title>([\s\S]*?)<\/title>/gi)].map((m) => decodeEntities(m[1].trim()));
    meta.manifestTitles = t;
    meta.manifestId = (man.match(/<manifest[^>]*\bidentifier="([^"]*)"/) || [])[1] || "";
  } catch (e) { meta.manifestTitles = []; }
  meta.title = meta.htmlTitle;
  meta.subtitle = meta.manifestTitles[1] || "";

  const reg = dataRegion(html);
  const sandbox = {};
  vm.createContext(sandbox);
  try { vm.runInContext(reg.src, sandbox, { timeout: 2000 }); }
  catch (e) { throw new Error("Không chạy được vùng dữ liệu (" + reg.how + "): " + e.message); }
  const NAMES = ["EXAM", "LOCK", "PARTS", "_KM", "_KX", "EXAM_INFO", "EXAM_LOCK", "TFNG", "PASSAGES", "EXERCISES", "ATOM_INFO", "TEACHER_PIN", "QUESTIONS"];
  const g = {};
  NAMES.forEach((n) => { if (sandbox[n] !== undefined) g[n] = clone(sandbox[n]); });
  const otherGlobals = Object.keys(sandbox).filter((k) => NAMES.indexOf(k) < 0);
  if (otherGlobals.length) warnings.push("Biến toàn cục khác trong vùng dữ liệu: " + otherGlobals.join(", "));

  const variant = detect(g);
  let KEYS = null, decStat = null;
  if (g._KX && g._KM) {
    KEYS = decryptKeys(g._KX, g._KM);
    decStat = { count: 0, used: new Set(), bad: [] };
    decryptAns(g.PARTS, KEYS, decStat);
    if (g.EXAM) decryptAns(g.EXAM, KEYS, decStat);
    if (decStat.bad.length) warnings.push("Chỉ mục đáp án ngoài bảng: " + decStat.bad.join(","));
    if (decStat.used.size !== KEYS.length) warnings.push("Bảng KEYS có " + KEYS.length + " mục nhưng chỉ dùng " + decStat.used.size);
  }
  // còn sót ans kiểu số?
  (function chk(n, p) {
    if (Array.isArray(n)) return n.forEach((x, i) => chk(x, p + "[" + i + "]"));
    if (n && typeof n === "object") for (const k of Object.keys(n)) {
      if (k === "ans" && typeof n[k] === "number") warnings.push("ans vẫn là số ở " + p + "." + k);
      chk(n[k], p + "." + k);
    }
  })(g, "");

  let norm;
  if (variant === "tpl-enc" || variant === "tpl-plain") {
    norm = { EXAM: clone(g.EXAM), PARTS: clone(g.PARTS), grading: {
      textMatch: "norm(): trim, lowercase, ‘’ʼ->', gộp khoảng trắng, bỏ [.,;:!?] cuối; loose(): bỏ '-' và khoảng trắng, '.'->','; bỏ mạo từ a/an/the đầu",
      tf: "EXAM.tfLadder[số ý đúng] × perItem", multi: "perItem × min(số chữ cái đúng, nums.length); không trừ chọn sai; chọn quá pick thì bỏ lựa chọn cũ nhất",
      pickBest: "nếu EXAM.pickBest: chỉ lấy take phần điểm cao nhất" } };
  } else if (variant === "ielts-v0") norm = normIelts(g, meta);
  else if (variant === "thpt-v0") norm = normThpt(g, meta);
  else if (variant === "k4-v0") norm = normK4(g, meta);
  else if (variant === "kid-adv") norm = normKidAdv(g, meta);
  else if (variant === "kid-math") {
    let vf = null;
    const src = sliceFunction(reg.script, "visualHTML");
    if (src) { const sb = {}; vm.createContext(sb); vm.runInContext(src, sb); vf = sb.visualHTML; }
    else warnings.push("Không thấy function visualHTML — mất hình minh hoạ dựng bằng mã");
    norm = normKidMath(g, meta, vf);
  } else throw new Error("Biến thể chưa biết; biến tìm thấy: " + Object.keys(g).join(","));

  // ảnh
  const refs = collectImages({ EXAM: norm.EXAM, PARTS: norm.PARTS });
  const files = listFiles(dir).filter((f) => /\.(png|jpe?g|gif|svg|webp)$/i.test(f));
  const referenced = [...refs.keys()];
  const images = {
    referenced: referenced.map((src) => ({ src, exists: fs.existsSync(path.join(dir, src)), usedAt: refs.get(src).slice(0, 4) })),
    missing: referenced.filter((s) => !/^(data:|https?:)/.test(s) && !fs.existsSync(path.join(dir, s))),
    unreferenced: files.filter((f) => referenced.indexOf(f) < 0),
  };

  return {
    source: path.basename(dir), variant, dataRegion: reg.how,
    encrypted: !!KEYS, nKeys: KEYS ? KEYS.length : 0, decryptedAnsFields: decStat ? decStat.count : 0,
    meta, EXAM: norm.EXAM, PARTS: norm.PARTS, images, grading: norm.grading,
    css: cssForContent(html, { EXAM: norm.EXAM, PARTS: norm.PARTS }),
    raw: redactPins(variant.indexOf("tpl") === 0 ? { LOCK: g.LOCK, _KM: g._KM } : g),
    warnings,
  };
}

/* ---------------------------------------------------------------- CLI */
if (require.main === module) {
  const a = process.argv.slice(2);
  if (a[0] === "--all") {
    const root = a[1];
    for (const d of fs.readdirSync(root)) {
      const fp = path.join(root, d);
      if (!fs.statSync(fp).isDirectory()) continue;
      try {
        const r = extract(fp);
        fs.writeFileSync(path.join(root, d + ".json"), JSON.stringify(r, null, 1));
        console.log("OK  ", d, r.variant, "enc=" + r.encrypted, "warn=" + r.warnings.length);
        r.warnings.forEach((w) => console.log("     !", w));
      } catch (e) { console.log("FAIL", d, e.message); }
    }
  } else if (a[0]) {
    const r = extract(a[0]);
    const s = JSON.stringify(r, null, 1);
    if (a[1]) fs.writeFileSync(a[1], s); else process.stdout.write(s);
  } else {
    console.error("node extract2.cjs <dir|index.html> [out.json]  |  node extract2.cjs --all <corpusDir>");
    process.exit(1);
  }
}
module.exports = { extract, decryptKeys, dataRegion };
